cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(CVIII_PC_Import_5H8I LANGUAGES C)
  enable_testing()
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../cviii_pc_world_5h8h" "${CMAKE_CURRENT_BINARY_DIR}/native-world")
endif()
add_library(cviii_pc_import_5h8i STATIC cviii_pc_import_5h8i.c)
target_link_libraries(cviii_pc_import_5h8i PUBLIC cviii_pc_world_5h8h)
target_include_directories(cviii_pc_import_5h8i PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_compile_features(cviii_pc_import_5h8i PUBLIC c_std_11)
if(MSVC)
  target_compile_options(cviii_pc_import_5h8i PRIVATE /W4 /utf-8)
else()
  target_compile_options(cviii_pc_import_5h8i PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(cviii_pc_import_test_5h8i cviii_pc_import_test_5h8i.c)
target_link_libraries(cviii_pc_import_test_5h8i PRIVATE cviii_pc_import_5h8i)
add_test(NAME CVIII_5H8I_PC_World_Import COMMAND cviii_pc_import_test_5h8i)
