cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(CVIII_PC_World_5H8H LANGUAGES C)
    enable_testing()
endif()
add_library(cviii_pc_world_5h8h STATIC cviii_pc_world.c cviii_pc_save.c)
target_include_directories(cviii_pc_world_5h8h PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(cviii_pc_world_5h8h PUBLIC c_std_11)
if(MSVC)
    target_compile_options(cviii_pc_world_5h8h PRIVATE /W4 /utf-8)
else()
    target_compile_options(cviii_pc_world_5h8h PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(cviii_pc_world_test_5h8h cviii_pc_world_test.c)
target_link_libraries(cviii_pc_world_test_5h8h PRIVATE cviii_pc_world_5h8h)
add_test(NAME CVIII_5H8H_PC_World_Foundation COMMAND cviii_pc_world_test_5h8h)
