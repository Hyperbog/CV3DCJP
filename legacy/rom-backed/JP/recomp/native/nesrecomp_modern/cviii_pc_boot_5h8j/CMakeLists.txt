cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(CVIII_PC_Boot_5H8J LANGUAGES C)
  enable_testing()
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../cviii_pc_world_5h8h" "${CMAKE_CURRENT_BINARY_DIR}/native-world")
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../cviii_pc_import_5h8i" "${CMAKE_CURRENT_BINARY_DIR}/native-import")
endif()
add_executable(cviii_pc_boot_5h8j cviii_pc_boot_5h8j.c)
target_link_libraries(cviii_pc_boot_5h8j PRIVATE cviii_pc_import_5h8i)
target_compile_features(cviii_pc_boot_5h8j PRIVATE c_std_11)
if(MSVC)
  target_compile_options(cviii_pc_boot_5h8j PRIVATE /W4 /utf-8)
else()
  target_compile_options(cviii_pc_boot_5h8j PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME CVIII_5H8J_ROM_Free_Bootstrap COMMAND
         "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/../../../tests/test_cviii_pc_boot_5h8j.py"
         "$<TARGET_FILE:cviii_pc_boot_5h8j>")
