cmake_minimum_required(VERSION 3.20)
project(CVIII_PersistentLevel_Policy C)
enable_testing()
add_subdirectory(../cviii_pc_world_5h8h "${CMAKE_CURRENT_BINARY_DIR}/world_5h8h")
add_library(cviii_pc_persistence_5h8v STATIC cviii_pc_persistence.c)
target_link_libraries(cviii_pc_persistence_5h8v PUBLIC cviii_pc_world_5h8h)
target_compile_features(cviii_pc_persistence_5h8v PUBLIC c_std_11)
if(MSVC)
  target_compile_options(cviii_pc_persistence_5h8v PRIVATE /W4 /utf-8)
else()
  target_compile_options(cviii_pc_persistence_5h8v PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(cviii_pc_persistence_test_5h8v cviii_pc_persistence_test.c)
target_link_libraries(cviii_pc_persistence_test_5h8v PRIVATE cviii_pc_persistence_5h8v)
add_test(NAME CVIII_5H8V_DestructiblesAndRespawn COMMAND cviii_pc_persistence_test_5h8v)
